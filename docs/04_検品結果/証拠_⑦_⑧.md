    {
        if (! preg_match('/^[0-9]{13}$/', $isbn)) {

        try {

            'title' => $volumeInfo['title'] ?? '',
            'author' => implode('、', $volumeInfo['authors'] ?? []),
            'description' => $volumeInfo['description'] ?? '',
            'image_url' => $volumeInfo['imageLinks']['thumbnail'] ?? '',
