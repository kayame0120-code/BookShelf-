# 検品表 11 データ

- 対象の発注書: `docs/02_発注書/発注書_11_データ.md`（この番号で何を作るかを書いた実装指示書）
- 共通の規則: `docs/共通ルール.md` の2章（証拠を集める条件・判定語を書かない原則・判定の主体・期待値の基準・確認方法の書き方・差し戻し）
- 材料: 全体仕様書 6章（テーブル一覧・外部キーと削除時の挙動・初期データの件数）・11章（全13テーブル・78カラムの定義）、技術決定書 TD-10（論理削除）・TD-11（ISBN・出版日をNULL可にする）・TD-19（初期データを乱数で決める）
- 補足: ルート・画面は持たない。テーブルはテーブル名とカラム名で確かめ、マイグレーションのファイル名では確かめない
- 確認方法と期待値に出てくる `$` で始まる名前は、下の「確認に使うもの」の準備で値を入れた変数である。期待値はその変数の値に置き換えて照合する
- 行の見方: 「確認方法」を実行して得た生の結果が「期待値」と一致すれば YES、一致しなければ NO と判定する。判定は共通ルール2-3に従い、証拠を集めたセッションの外で行う

## 確認に使うもの

- テーブル定義の照合は、全体仕様書 11章（`docs/01_仕様書/全体仕様書.md`）の表から作った一覧と、実際のDBの `information_schema` から作った一覧を `diff` で突き合わせる
- 外部キーの削除時の挙動は、トランザクションの中で削除を試し、最後に必ず巻き戻す。DBの状態は変わらない
- 初期データのうち、レビュー（`reviews`）といいね（`review_likes`）は書籍の登録者・レビューの投稿者・件数・評価を乱数で決めるため、実行のたびに件数が変わる。この2つは固定の件数で一致を求めず、それ以外のテーブルだけを固定の件数と、2回の投入での一致で確かめる
- 「初期データの投入を2回」は、`sail artisan migrate:fresh --seed` を2回実行することを指す
- JSON・表の加工には `python3` を使う

export LC_ALL=C.UTF-8
BASE=http://localhost            # アプリのURL（Sail の既定値）
STAMP=$(date +%s)
G=/tmp/jarG; rm -f $G            # 一度もログインしない利用者の cookie 保存先
tk()    { sail artisan tinker --execute="$1"; }
uid()   { tk "echo App\Models\User::where('email','$1$STAMP@example.com')->value('id');"; }

login_as() { curl -s -o /dev/null -c "$1" -b "$1" -X POST "$BASE/register" \
  --data-urlencode "_token=$(token_from "$1" /register)" --data-urlencode "name=検品会員$2" \
  --data-urlencode "email=$2$STAMP@example.com" --data-urlencode "password=password123" \
  --data-urlencode "password_confirmation=password123"; }
page()  { curl -s -c "$1" -b "$1" "$BASE$2"; }
has()   { page "$1" "$2" | grep -cF -- "$3"; }            # 本文に $3 を含む行の数
near()  { page "$1" "$2" | grep -B3 -A3 -F -- "$3"; }      # $3 の前後3行
st()    { curl -s -o /dev/null -w '%{http_code} %{redirect_url}\n' -c "$1" -b "$1" "$BASE$2"; }
# submit <cookie> <トークンを取る画面> <POST|PUT|PATCH|DELETE> <送信先> [項目=値 ...]
# 送信結果を "STATUS=… LOCATION=…" で出力し、移動先（移動が無ければ応答そのもの）の本文を /tmp/last.html に保存する
submit() {
  local jar=$1 form=$2 method=$3 action=$4; shift 4
  local args=(--data-urlencode "_token=$(token_from "$jar" "$form")" --data-urlencode "_method=$method") kv
  for kv in "$@"; do args+=(--data-urlencode "$kv"); done
  local res; res=$(curl -s -o /tmp/resp.html -w '%{http_code} %{redirect_url}' -c "$jar" -b "$jar" -e "$BASE$form" -X POST "$BASE$action" "${args[@]}")
  local loc=${res#* }
  echo "STATUS=${res%% *} LOCATION=$loc"
  if [ -n "$loc" ]; then curl -s -c "$jar" -b "$jar" "$loc" > /tmp/last.html; else cp /tmp/resp.html /tmp/last.html; fi
}

```bash
SPEC=docs/01_仕様書/全体仕様書.md
# spec_cols — 全体仕様書 11章の各カラムを「テーブル カラム 型 NULL可否」の1行ずつで出す
spec_cols() { python3 - "$SPEC" <<'PY'
import re,sys
t=None; on=False
for l in open(sys.argv[1],encoding='utf-8'):
    if l.startswith('## 11.'): on=True
    if not on: continue
    m=re.match(r'### (\w+)（',l)
    if m: t=m.group(1); continue
    c=[x.strip() for x in l.strip().strip('|').split('|')]
    if t and len(c)>=7 and c[0] not in ('カラム','---'):
        print(t, c[0], c[1], 'YES' if c[2]=='可' else 'NO')
PY
}
db_cols() { tk 'foreach (DB::select("select table_name t, column_name c, column_type ty, is_nullable n from information_schema.columns where table_schema = database() order by table_name, ordinal_position") as $r) echo "$r->t $r->c $r->ty $r->n\n";' | sort; }
# spec_fks — 全体仕様書 11章の外部キーを「テーブル カラム 参照先 削除時」で出す
spec_fks() { python3 - "$SPEC" <<'PY'
import re,sys
t=None; on=False
for l in open(sys.argv[1],encoding='utf-8'):
    if l.startswith('## 11.'): on=True
    if not on: continue
    m=re.match(r'### (\w+)（',l)
    if m: t=m.group(1); continue
    c=[x.strip() for x in l.strip().strip('|').split('|')]
    if t and len(c)>=7:
        m=re.match(r'(\w+\.\w+)・(restrict|cascade)',c[5])
        if m: print(t, c[0], m.group(1), m.group(2).upper())
PY
}
db_fks() { tk 'foreach (DB::select("select k.table_name t, k.column_name c, concat(k.referenced_table_name, \".\", k.referenced_column_name) r, rc.delete_rule d from information_schema.key_column_usage k join information_schema.referential_constraints rc on rc.constraint_name = k.constraint_name and rc.constraint_schema = k.constraint_schema where k.table_schema = database() and k.referenced_table_name is not null") as $r) echo "$r->t $r->c $r->r $r->d\n";' | sort; }
counts() { tk 'foreach (["users","genres","books","book_genre","favorites","reading_plans","notifications","personal_access_tokens","password_reset_tokens","failed_jobs","migrations"] as $t) echo $t, " ", DB::table($t)->count(), "\n";'; }
# fixed_hash — 乱数を使わないテーブルの中身（日時の列を除く）を1つの値にまとめる
fixed_hash() { tk 'echo md5(json_encode([DB::table("users")->orderBy("id")->get(["id","name","email"]), DB::table("genres")->orderBy("id")->get(["id","name"]), DB::table("books")->orderBy("id")->get(["id","title","author","isbn","published_date","description","image_url"]), DB::table("book_genre")->orderBy("book_id")->orderBy("genre_id")->get(), DB::table("favorites")->orderBy("user_id")->orderBy("book_id")->get(), DB::table("reading_plans")->orderBy("id")->get(["id","user_id","book_id","status"])]));'; }
# try_delete <SQLのwhereまで含む削除文> — トランザクションの中で削除を試し、巻き戻す。拒否されたら REFUSED、通れば DELETED と出す
try_delete() { tk "DB::beginTransaction(); try { DB::delete(\"$1\"); echo 'DELETED'; } catch (\Throwable \$e) { echo 'REFUSED'; } DB::rollBack();"; }
# after_delete <削除文> <確かめるSQL> — トランザクションの中で削除し、確かめるSQLの件数を出して巻き戻す
after_delete() { tk "DB::beginTransaction(); DB::delete(\"$1\"); echo DB::selectOne(\"$2\")->c; DB::rollBack();"; }
```
